int fill_rectangle(
    HDC handle_to_device_context, 
    const RECTANGLE& rectangle
)

This function is used to fill in a rectangle.

The parameter {handle_to_device_context} is a handle to the device context.
The parameter {rectangle} is the rectangle to fill in.

Normally returns 0 but returns 1 if there is no brush in {rectangle}.



inline int update_brush(
    RECTANGLE& rectangle
)

This function is used to update the rectangle's brush and handles its memory.

The parameter {rectangle} is the rectangle with the brush and color.

Normally returns 0.



int update_rectangle(
    RECTANGLE &rectangle
)

This function is used to update and sync the rectangle with the RECT structure within it.

The parameter {rectangle} is the rectangle to update

Uses update_brush.

Normally returns 0.



int draw_rectangles(
    HDC handle_device_context, 
    std::vector<RECTANGLE>* rectangles
)

This function is used to draw rectangles on a window.

The parameter {handle_device_context} is a handle to the device context.
The parameter {rectangles} are the rectangles to be drawn.

Uses update_rectangle.

Normally returns 0 but returns 1 if {rectangles} is empty.



/// window_procedure.hpp ///



int update_previous_window_dimensions(
    int new_width, 
    int new_height, 
    WINDOW_PROCEDURE_DATA& window_procedure_data
)

This function is used to update the previous window dimensions within {window_procedure_data}.

The parameter {new_width} and {new_height} are the new dimensions used to update.
The parameter {window_procedure_data} contains the previous window dimensions to be updated.

Normally returns 0 but returns 1 if {new_width} or {new_height} is invalid.

/// Usage ///



// so to start, include BracketsGF/brackets.hpp:


#pragma comment(lib, "user32.lib")
#pragma comment(lib, "gdi32.lib")
#pragma comment(lib, "kernel32.lib")

#include "BracketsGF/brackets.hpp"


// then you initialize a WinMain function:


int WINAPI WinMain(
    HINSTANCE instance_handle, 
    HINSTANCE handle_to_previous, 
    LPSTR long_pointer_to_command, 
    int show_window_command
) { // ...


// then you initialize your window and the following parameters:


WINDOW window = {};
window.window_title = L"window title";
window.window_class = L"window class";
window.rectangles = std::make_unique<std::vector<RECTANGLE>>(); // required

window.window_procedure_data.previous_window_width = 800; // required base width
window.window_procedure_data.previous_window_height = 600; // required base height


// then you initialize your window class:


WNDCLASSW window_class = {};
window_class.lpfnWndProc = window_procedure; // window_procedure is a function
window_class.hInstance = instance_handle;
window_class.lpszClassName = window.window_class;
window_class.hCursor = LoadCursorW(NULL, (LPCWSTR)IDC_ARROW);
RegisterClassW(&window_class);


// then you need to initialize your window.window_handle:


window.window_handle = CreateWindowExW(
    0, window.window_class, window.window_title,
    WS_OVERLAPPEDWINDOW | WS_VISIBLE, 
    CW_USEDEFAULT, CW_USEDEFAULT, 800, 600, 
    nullptr, nullptr, instance_handle, &window
);


// then you can initialize your rectangles like this:


RECTANGLE green_square = {};
green_square.base_x = 100.0;
green_square.base_y = 100.0;
green_square.base_width = 150;
green_square.base_height = 150;

green_square.x = green_square.base_x;
green_square.y = green_square.base_y;
green_square.width = green_square.base_width;
green_square.height = green_square.base_height;

green_square.color = RGB(0, 255, 0);

update_rectangle(green_square); // sync
add_rectangle(&green_square, window);


// then finally run the message loop:


    return run_message_loop();
}
